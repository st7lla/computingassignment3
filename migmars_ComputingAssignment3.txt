## Constants
# Board layout, the codes that move() returns, and the settings used for reading and displaying the board

P1_STORAGE = 0  # player 1's storage is always at index 0
ILLEGAL_MOVE = -1
GO_AGAIN = 1
TURN_OVER = 0

DIGITS = "0123456789"
CELL_WIDTH = 4


def import_board(filename):
    """Reads a Mancala board from a text file with one integer per line.
    Parameters include: filename - the name of the file to read. Returns the board as a list of integers if it is legal (even number of items, all integers greater than or equal to 0), otherwise returns False, and it's output is nothing"""

    board = []
    file = open(filename)
    for line in file:
        text = line.strip()  # removes the newline at the end of the line
        if text != "":
            # every character has to be a digit, so negatives and decimals fail
            for character in text:
                if character not in DIGITS:
                    file.close()
                    return False
            board.append(int(text))
    file.close()

    num_slots = len(board)
    if num_slots < 2 or num_slots % 2 != 0:
        return False
    return board

## Display Functions

def pad(value):
    """Adds spaces after a value so that it takes up CELL_WIDTH characters.
    Parameters include: value - an integer or string to convert and pad. Returns the padded string, and its output is nothing"""

    text = str(value)
    while len(text) < CELL_WIDTH:
        text = text + " "
    return text


def show_board(board):
    """Displays a Mancala board in the shell with the storages on the sides.
    Parameters include: board - the list representing the board. Does not return anything, and its output is three lines of text, with player 1's wells on top and player 2's wells on the bottom"""

    num_slots = len(board)
    p2_storage = num_slots // 2

    # top row: player 1's wells, left to right
    print(pad(""), end="")
    for i in range(1, p2_storage):
        print(pad(board[i]), end="")
    print()

    # middle row: both storages with space between them
    print(pad(board[P1_STORAGE]), end="")
    for i in range(1, p2_storage):
        print(pad(""), end="")
    print(board[p2_storage])

    # bottom row: player 2's wells are printed from the highest index down
    print(pad(""), end="")
    for i in range(num_slots - 1, p2_storage, -1):
        print(pad(board[i]), end="")
    print()


## Scoring Function

def score(board):
    """Calculates the score of each player (tokens in their wells plus their storage).
    Parameters include: board - the list representing the board. Returns two integers, player 1's score followed by player 2's score, and its output is nothing"""

    num_slots = len(board)
    p2_storage = num_slots // 2

    # accumulator pattern: running totals for each player
    p1_score = 0
    p2_score = 0
    for i in range(num_slots):
        if i < p2_storage:
            p1_score = p1_score + board[i]
        else:
            p2_score = p2_score + board[i]
    return p1_score, p2_score


## Move Function

def move(board, index):
    """Processes one move on the board without changing the original list.
    Parameters include: board - the list representing the board and index - the well the player picked. Returns the new board and a code: 1 if the player lands in their own storage and goes again, 0 if their turn ends, or -1 (with the original board) if the move is illegal, and its output is nothing"""

    num_slots = len(board)
    p2_storage = num_slots // 2

    # a storage, an empty well, or an index off the board is not a legal move
    if (index <= 0 or index >= num_slots or index == p2_storage
            or board[index] == 0):
        return board, ILLEGAL_MOVE

    # the index tells us which player is moving
    if index < p2_storage:
        my_storage = P1_STORAGE
        opp_storage = p2_storage
        first_well = 1
        last_well = p2_storage - 1
    else:
        my_storage = p2_storage
        opp_storage = P1_STORAGE
        first_well = p2_storage + 1
        last_well = num_slots - 1

    # copy of the board so the original list is never changed
    new_board = []
    for tokens_in_slot in board:
        new_board.append(tokens_in_slot)

    tokens = new_board[index]
    new_board[index] = 0

    # counter-clockwise is decreasing index, wrapping around, skipping the opponent's storage
    position = index
    while tokens > 0:
        position = (position - 1) % num_slots
        if position != opp_storage:
            new_board[position] = new_board[position] + 1
            tokens = tokens - 1

    if position == my_storage:
        return new_board, GO_AGAIN

    # a well with exactly 1 token was empty before, so capture the opposite well
    if first_well <= position and position <= last_well:
        if new_board[position] == 1:
            opposite = num_slots - position
            new_board[my_storage] = new_board[my_storage] + new_board[opposite]
            new_board[opposite] = 0

    return new_board, TURN_OVER
